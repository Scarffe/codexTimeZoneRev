cmake_minimum_required(VERSION 3.14)
project(window_chrome_tests LANGUAGES CXX)
set(RUNNER ../../flutter_app/windows/runner)
add_executable(window_chrome_tests tests.cpp ${RUNNER}/window_chrome.cpp ${RUNNER}/Runner.rc)
target_include_directories(window_chrome_tests PRIVATE ${RUNNER})
target_compile_features(window_chrome_tests PRIVATE cxx_std_17)
target_compile_options(window_chrome_tests PRIVATE /W4 /WX /utf-8 /EHsc)
target_compile_definitions(window_chrome_tests PRIVATE NOMINMAX UNICODE _UNICODE)
target_link_libraries(window_chrome_tests PRIVATE dwmapi ole32 shell32 user32)
enable_testing()
add_test(NAME window_chrome_contract COMMAND window_chrome_tests)
