cmake_minimum_required(VERSION 3.21)
project(CodexRuntimeBridge LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 runtime bridge only")
endif()
add_library(codex_runtime_bridge SHARED runtime_bridge.cpp)
add_executable(codex_runtime_metadata runtime_metadata.cpp)
target_compile_features(codex_runtime_bridge PRIVATE cxx_std_17)
target_compile_features(codex_runtime_metadata PRIVATE cxx_std_17)
target_compile_definitions(codex_runtime_bridge PRIVATE UNICODE _UNICODE)
target_compile_definitions(codex_runtime_metadata PRIVATE UNICODE _UNICODE)
target_link_libraries(codex_runtime_bridge PRIVATE bcrypt shell32)
option(CTZ_BRIDGE_OWNED_TEST "Owned Node fixture ABI export" OFF)
if(CTZ_BRIDGE_OWNED_TEST)
  target_compile_definitions(codex_runtime_bridge PRIVATE CTZ_BRIDGE_OWNED_TEST)
endif()
