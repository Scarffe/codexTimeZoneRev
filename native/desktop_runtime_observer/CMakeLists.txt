cmake_minimum_required(VERSION 3.21)
project(ctz_desktop_runtime_observer LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The Desktop runtime observer requires Windows x64")
endif()
add_library(codex_desktop_observer SHARED observer_dll.cpp)
target_compile_features(codex_desktop_observer PRIVATE cxx_std_17)
target_compile_definitions(codex_desktop_observer PRIVATE UNICODE _UNICODE)
target_link_libraries(codex_desktop_observer PRIVATE bcrypt)
