cmake_minimum_required(VERSION 3.21)
project(ctz_desktop_runtime_observer LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The Desktop runtime observer requires Windows x64")
endif()
add_library(codex_desktop_observer SHARED observer_dll.cpp)
target_compile_features(codex_desktop_observer PRIVATE cxx_std_17)
target_compile_definitions(codex_desktop_observer PRIVATE UNICODE _UNICODE)
target_link_libraries(codex_desktop_observer PRIVATE bcrypt)

option(CTZ_BUILD_OBSERVER_TESTS "Build CPU-only production projection contracts" OFF)
if(CTZ_BUILD_OBSERVER_TESTS)
  enable_testing()
  foreach(contract IN ITEMS v4_projection_limits v4_projection_selected_boundaries v4_bootstrap)
    add_executable(${contract}_contract ../../validation/global_monitor/${contract}_contract.cpp)
    target_compile_features(${contract}_contract PRIVATE cxx_std_17)
    target_compile_definitions(${contract}_contract PRIVATE UNICODE _UNICODE)
    add_test(NAME ${contract} COMMAND ${contract}_contract)
  endforeach()
endif()
